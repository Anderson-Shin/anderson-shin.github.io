# 5. VaR, ES 계산 (95%)
alpha = 0.95
VaR_daily = -np.percentile(portfolio_returns, 100 * (1 - alpha))
es_daily = -portfolio_returns[portfolio_returns <= -VaR_daily].mean()

# 6. SRM 계산 함수
def phi_gamma(p, gamma):
    return np.exp(-(1 - p) / gamma) / (gamma * (1 - np.exp(-1 / gamma)))

def compute_srm(returns, gamma):
    sorted_losses = -np.sort(returns)
    n = len(sorted_losses)
    p = np.arange(1, n + 1) / n
    weights = phi_gamma(p, gamma)
    weights /= weights.sum()
    return np.sum(weights * sorted_losses)

# 감마 = 1.0 기준 SRM
srm_daily = compute_srm(portfolio_returns.values, gamma=1.0)

# 7. 결과 요약 출력
summary = {
    "Mean Return": mean_daily,
    "Standard Deviation": std_daily,
    "Variance": var_daily,
    "VaR (95%)": VaR_daily,
    "Expected Shortfall (95%)": es_daily,
    "SRM (gamma=1.0)": srm_daily
}

summary_df = pd.DataFrame.from_dict(summary, orient='index', columns=['Value'])
print("=== Portfolio Risk Summary ===")
print(summary_df)

```python
# 감마 기반 phi 함수
def phi_gamma(p, gamma):
    return np.exp(-(1 - p) / gamma) / (gamma * (1 - np.exp(-1 / gamma)))

# SRM 계산 함수
def compute_srm(returns, gamma):
    sorted_losses = -np.sort(returns)  # 손실 정렬
    n = len(sorted_losses)
    p = np.arange(1, n + 1) / n
    weights = phi_gamma(p, gamma)
    weights /= weights.sum()  # 정규화
    return np.sum(weights * sorted_losses)

# 감내 가능한 손실 수준에 맞는 감마를 찾는 함수
def find_gamma_for_risk_appetite(returns, annual_threshold=-0.20, gamma_min=0.1, gamma_max=5.0, steps=100, trading_days=252):
    gamma_range = np.linspace(gamma_min, gamma_max, steps)
    for gamma in gamma_range:
        srm_daily = compute_srm(returns, gamma)
        srm_annual = srm_daily * np.sqrt(trading_days)
        if srm_annual >= annual_threshold:
            return gamma, srm_daily, srm_annual
    return None, None, None

# 사용 예시: 감내 가능한 손실 -20% 기준
risk_threshold = -0.20
best_gamma, srm_daily, srm_annual = find_gamma_for_risk_appetite(portfolio_returns.values, annual_threshold=risk_threshold)

# 출력
if best_gamma:
    print(f"▶ 감내 가능한 손실 -20% 기준 적정 감마값: γ = {best_gamma:.2f}")
    print(f"  - SRM (Daily): {srm_daily:.4f}")
    print(f"  - SRM (Annualized): {srm_annual:.4f}")
else:
    print(f"⚠ 감내 가능한 수준인 {risk_threshold*100:.1f}%을 만족하는 감마 값이 존재하지 않습니다.")
```

# 15-Advanced: 평균 수익률 극대화 + SRM ≤ –20% 제약 동시 최적화
from scipy.optimize import differential_evolution
import numpy as np

target_annual = -0.20
trading_days  = 252

# 목적함수: 평균수익 최대화, SRM 제약 위반 시 페널티 부과
def obj_mean_max(x):
    γ     = x[0]
    w_raw = x[1:]
    w     = w_raw / np.sum(w_raw)                      # 합 = 1
    pr    = (returns.values * w).sum(axis=1)           # 일별 포트폴리오 수익률
    mean_a= pr.mean() * trading_days                   # 연환산 평균 수익률
    srm_a = compute_srm(pr, γ) * np.sqrt(trading_days) # 연환산 SRM
    # SRM이 |–20%| 초과 시 큰 페널티
    penalty = 0
    if srm_a > abs(target_annual):
        penalty = 1e3 * (srm_a - abs(target_annual))**2
    return -mean_a + penalty

# γ ∈ [0.1,5], w_rawᵢ ∈ [0,1]
bounds = [(0.1, 5.0)] + [(0.0, 1.0)] * len(tickers)

# DE 실행 (재현성 위해 seed 고정, disp=False로 로그 숨김)
result = differential_evolution(
    obj_mean_max,
    bounds,
    maxiter=500,
    popsize=15,
    tol=1e-3,
    disp=False,
    polish=True,
    seed=42
)

# 최적 해 추출
γ_opt        = result.x[0]
w_opt        = result.x[1:] / np.sum(result.x[1:])
port_ret_opt = (returns.values * w_opt).sum(axis=1)

# 일별 리스크·수익 지표
mean_d = port_ret_opt.mean()
std_d  = port_ret_opt.std()
VaR_d  = -np.percentile(port_ret_opt, 100 * (1 - alpha))
ES_d   = -port_ret_opt[port_ret_opt <= -VaR_d].mean()
SRM_d  = compute_srm(port_ret_opt, γ_opt)

# 연환산 지표
mean_a = mean_d * trading_days
std_a  = std_d * np.sqrt(trading_days)
VaR_a  = VaR_d * np.sqrt(trading_days)
ES_a   = ES_d * np.sqrt(trading_days)
SRM_a  = SRM_d * np.sqrt(trading_days)

# 결과 출력
print(f"▶ 감마 {γ_opt:.2f} 기준, SRM ≤ {int(target_annual*100)}% 제약하에 평균 수익률 최대화 포트폴리오:")
for t, w in zip(tickers, w_opt):
    print(f"  - {t}: {w:.4f}")

print(f"""
[Daily Metrics]
